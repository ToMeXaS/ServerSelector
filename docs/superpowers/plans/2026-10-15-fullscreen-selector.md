# Fullscreen Server Selector Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the bossbar HUD with a fullscreen, mouse-driven carousel server selector built from per-player display entities and Nexo glyphs, with live server status and an ajQueue queue panel.

**Architecture:** Each player rides an invisible pig that is also their camera, so their head rotation keeps reaching the server and is projected onto a UI plane as a cursor. A `SelectorSession` owns the seat, the cursor and two screens (`CarouselScreen`, `QueuePanel`) made of `UiElement`s (text displays positioned by their centre on a `UiPlane`). `ServerRegistry` holds config-defined servers plus live status from ajQueue and player counts from the proxy. Pure math and parsing are unit tested. Everything that needs a running server is verified in game on the test network.

**Tech Stack:** Java 21, Paper API 1.21.4 (running on Paper 26.3), ProtocolLib 5.4.0, ajQueue API 2.9.1, Adventure, JUnit 5, Python 3.11 + Pillow (asset build), Nexo 1.30-dev (resource pack host).

**Spec:** `docs/superpowers/specs/2026-10-15-fullscreen-selector-design.md`

## Global Constraints

- Java toolchain 21. Compile against `io.papermc.paper:paper-api:1.21.4-R0.1-SNAPSHOT`. The test server runs Paper 26.3.
- No compile dependency on Nexo. Glyph characters are fixed in `nexo/glyphs/selector.yml`, chars `\uE300`–`\uE33F`, written as `\uXXXX` escapes.

```python
#!/usr/bin/env python3
"""Builds the server selector's Nexo glyphs from Kenney's CC0 asset packs.

Run from anywhere:  python tools/build_selector_assets.py
Needs Pillow:       python -m pip install -r tools/requirements.txt

Writes:
  nexo/glyphs/selector.yml
  nexo/pack/assets/minecraft/textures/selector/*.png
  nexo/LICENSE-kenney.txt
Regenerate rather than editing the outputs: the plugin's Glyph enum and layout assume these
characters and sizes.
"""
import io
import urllib.request
import zipfile
from pathlib import Path

from PIL import Image, ImageDraw

ROOT = Path(__file__).resolve().parent.parent
NEXO = ROOT / "nexo"
TEXTURES = NEXO / "pack" / "assets" / "minecraft" / "textures" / "selector"
CACHE = ROOT / "tools" / ".cache"

PACKS = {
    "cursor": "https://kenney.nl/media/pages/assets/cursor-pack/461b29df18-1717599281/kenney_cursor-pack.zip",
    "ui": "https://kenney.nl/media/pages/assets/ui-pack/f651646eab-1718203990/kenney_ui-pack.zip",
    "icons": "https://kenney.nl/media/pages/assets/game-icons/1ebf9c14af-1677661579/kenney_game-icons.zip",
}


def load_pack(name):
    CACHE.mkdir(parents=True, exist_ok=True)
    path = CACHE / f"{name}.zip"
    if not path.exists():
        request = urllib.request.Request(PACKS[name], headers={"User-Agent": "Mozilla/5.0"})
        with urllib.request.urlopen(request) as response:
            path.write_bytes(response.read())
    return zipfile.ZipFile(path)


def kenney(pack, member):
    with load_pack(pack) as archive:
        return Image.open(io.BytesIO(archive.read(member))).convert("RGBA")


def panel(width, height, fill, border, border_width, radius):
    image = Image.new("RGBA", (width, height), (0, 0, 0, 0))
    ImageDraw.Draw(image).rounded_rectangle(
        (0, 0, width - 1, height - 1), radius=radius, fill=fill, outline=border, width=border_width)
    return image


def solid(width, height, color):
    return Image.new("RGBA", (width, height), color)


def vignette(width, height):
    # Light in the middle so the lobby build shows through, darker towards the edges
    image = Image.new("RGBA", (width, height))
    pixels = image.load()
    for y in range(height):
        for x in range(width):
            dx = (x - width / 2) / (width / 2)
            dy = (y - height / 2) / (height / 2)
            d = min(1.0, (dx * dx + dy * dy) ** 0.5 / 1.2)
            pixels[x, y] = (0, 0, 0, int(60 + 150 * d * d))
    return image


# (name, char code, rendered height in px, image factory)
GLYPHS = [
    ("cursor_arrow", 0xE300, 16, lambda: kenney("cursor", "PNG/Outline/Default/pointer_b.png")),
    ("cursor_hand", 0xE301, 16, lambda: kenney("cursor", "PNG/Outline/Default/hand_point.png")),
    ("arrow_left", 0xE302, 24, lambda: kenney("ui", "PNG/Yellow/Default/arrow_decorative_w.png")),
    ("arrow_right", 0xE303, 24, lambda: kenney("ui", "PNG/Yellow/Default/arrow_decorative_e.png")),
    ("card", 0xE304, 128, lambda: panel(104, 128, (20, 24, 34, 200), (58, 67, 86, 255), 2, 8)),
    ("card_selected", 0xE305, 128, lambda: panel(104, 128, (40, 32, 20, 215), (255, 180, 58, 255), 3, 8)),
    ("panel", 0xE306, 140, lambda: panel(180, 140, (20, 26, 38, 235), (255, 180, 58, 255), 2, 8)),
    ("dim", 0xE307, 36, lambda: solid(64, 36, (0, 0, 0, 150))),
    ("vignette", 0xE308, 144, lambda: vignette(256, 144)),
    ("button", 0xE309, 32, lambda: kenney("ui", "PNG/Red/Default/button_rectangle_depth_flat.png")),
    ("button_hover", 0xE30A, 32, lambda: kenney("ui", "PNG/Red/Default/button_rectangle_depth_gradient.png")),
    ("bar_track", 0xE30B, 8, lambda: kenney("ui", "PNG/Grey/Default/slide_horizontal_grey.png")),
    ("bar_fill", 0xE30C, 8, lambda: kenney("ui", "PNG/Yellow/Default/slide_horizontal_color.png")),
    ("icon_sword", 0xE310, 32, lambda: kenney("cursor", "PNG/Outline/Default/tool_sword_a.png")),
    ("icon_pickaxe", 0xE311, 32, lambda: kenney("cursor", "PNG/Outline/Default/tool_pickaxe.png")),
    ("icon_axe", 0xE312, 32, lambda: kenney("cursor", "PNG/Outline/Default/tool_axe.png")),
    ("icon_bow", 0xE313, 32, lambda: kenney("cursor", "PNG/Outline/Default/tool_bow.png")),
    ("icon_hammer", 0xE314, 32, lambda: kenney("cursor", "PNG/Outline/Default/tool_hammer.png")),
    ("icon_wand", 0xE315, 32, lambda: kenney("cursor", "PNG/Outline/Default/tool_wand.png")),
    ("icon_bomb", 0xE316, 32, lambda: kenney("cursor", "PNG/Outline/Default/tool_bomb.png")),
    ("icon_torch", 0xE317, 32, lambda: kenney("cursor", "PNG/Outline/Default/tool_torch.png")),
    ("icon_star", 0xE318, 32, lambda: kenney("icons", "PNG/White/2x/star.png")),
]


def config_text():
    lines = [
        "# Generated by tools/build_selector_assets.py - edit the script, not this file.",
        "#",
        "# ascent = height / 2 + 3 centres every glyph on the text line, so the plugin can",
        "# position glyphs and text the same way. char is a \\uXXXX escape because Nexo reads",
        "# this file with the platform charset, which mangles literal private-use characters.",
        "",
    ]
    for name, code, height, _ in GLYPHS:
        lines += [
            f"selector_{name}:",
            f"  texture: selector/{name}",
            f"  ascent: {height // 2 + 3}",
            f"  height: {height}",
            f'  char: "\\u{code:04X}"',
            "",
        ]
    return "\n".join(lines)


def main():
    TEXTURES.mkdir(parents=True, exist_ok=True)
    for name, _, _, make in GLYPHS:
        make().save(TEXTURES / f"{name}.png")
    (NEXO / "glyphs").mkdir(parents=True, exist_ok=True)
    (NEXO / "glyphs" / "selector.yml").write_text(config_text(), encoding="utf-8", newline="\n")
    with load_pack("cursor") as archive:
        (NEXO / "LICENSE-kenney.txt").write_bytes(archive.read("License.txt"))
    print(f"Wrote {len(GLYPHS)} glyphs to {NEXO}")


if __name__ == "__main__":
    main()
```

```java
package lt.tomexas.serverselector.selector.ui;

/** Pure helpers for turning head rotation into a cursor position on the UI plane. */
public final class UiMath {

    /** Distance from the camera to the plane all UI coordinates are given on, in blocks. */
    public static final double REFERENCE_DISTANCE = 2.0;
    /** One font pixel of a text display at scale 1, in blocks. */
    public static final double PIXEL = 1.0 / 40.0;
    // tan() runs off to infinity near 90 degrees
    private static final double MAX_ANGLE = 80.0;

    private UiMath() {
    }

    /** Wraps an angle difference into [-180, 180). */
    public static double wrapDegrees(double degrees) {
        double wrapped = (degrees + 180.0) % 360.0;
        if (wrapped < 0) wrapped += 360.0;
        return wrapped - 180.0;
    }

    /** Where a line of sight turned by the given angle hits the reference plane. */
    public static double project(double degrees) {
        return REFERENCE_DISTANCE * Math.tan(Math.toRadians(clamp(degrees, -MAX_ANGLE, MAX_ANGLE)));
    }

    public static double clamp(double value, double min, double max) {
        return Math.max(min, Math.min(max, value));
    }
}
```

```java
package lt.tomexas.serverselector.selector.screen;

/** Where each server card sits in the carousel, relative to the selected one. */
public final class CarouselLayout {

    public enum Slot { CENTER, LEFT, RIGHT, HIDDEN_LEFT, HIDDEN_RIGHT }

    public static final double SIDE_X = 1.15;
    public static final double HIDDEN_X = 1.8;
    public static final float SIDE_SCALE = 0.7f;
    public static final float HIDDEN_SCALE = 0.4f;

    private CarouselLayout() {
    }

    /** Signed steps from the selected card, wrapped into (-count/2, count/2]. */
    public static int offset(int index, int selected, int count) {
        int offset = Math.floorMod(index - selected, count);
        if (offset > count / 2) offset -= count;
        return offset;
    }

    public static Slot slot(int index, int selected, int count) {
        int offset = offset(index, selected, count);
        if (offset == 0) return Slot.CENTER;
        if (offset == -1) return Slot.LEFT;
        if (offset == 1) return Slot.RIGHT;
        return offset < 0 ? Slot.HIDDEN_LEFT : Slot.HIDDEN_RIGHT;
    }

    public static double x(Slot slot) {
        return switch (slot) {
            case CENTER -> 0.0;
            case LEFT -> -SIDE_X;
            case RIGHT -> SIDE_X;
            case HIDDEN_LEFT -> -HIDDEN_X;
            case HIDDEN_RIGHT -> HIDDEN_X;
        };
    }

    public static float scale(Slot slot) {
        return switch (slot) {
            case CENTER -> 1.0f;
            case LEFT, RIGHT -> SIDE_SCALE;
            case HIDDEN_LEFT, HIDDEN_RIGHT -> HIDDEN_SCALE;
        };
    }

    public static boolean visible(Slot slot) {
        return slot != Slot.HIDDEN_LEFT && slot != Slot.HIDDEN_RIGHT;
    }

    public static int wrap(int index, int count) {
        return Math.floorMod(index, count);
    }
}
```

```java
package lt.tomexas.serverselector.server;

import java.util.Locale;
import java.util.Set;

/** A server's availability, as reported by ajQueue's status placeholder. */
public enum ServerStatus {
    ONLINE(true),
    // Full, whitelisted, paused or restricted: ajQueue still queues players for these
    LIMITED(true),
    OFFLINE(false),
    UNKNOWN(false);

    private static final Set<String> LIMITED_LABELS = Set.of("full", "whitelisted", "paused", "restricted");

    private final boolean joinable;

    ServerStatus(boolean joinable) {
        this.joinable = joinable;
    }

    public boolean joinable() {
        return joinable;
    }

    /** Maps ajQueue's status text (e.g. "&aOnline") to a status. Anything unrecognised counts as offline. */
    public static ServerStatus fromAjQueue(String raw) {
        String plain = plainText(raw).toLowerCase(Locale.ROOT);
        if (plain.isEmpty()) return UNKNOWN;
        if (plain.equals("online")) return ONLINE;
        if (LIMITED_LABELS.contains(plain)) return LIMITED;
        return OFFLINE;
    }

    /** Strips legacy colour codes and MiniMessage tags. */
    public static String plainText(String raw) {
        if (raw == null) return "";
        return raw.replaceAll("(?i)[&\u00A7][0-9a-fk-orx]", "").replaceAll("<[^>]+>", "").trim();
    }
}
```

```java
public record SelectorConfig(String title, List<ServerEntry> servers) {

    private static final String DEFAULT_TITLE = "CHOOSE YOUR REALM";

    public static SelectorConfig from(ConfigurationSection root, Logger logger) {
        List<ServerEntry> servers = new ArrayList<>();
        ConfigurationSection section = root.getConfigurationSection("servers");
        if (section != null) {
            for (String id : section.getKeys(false)) {
                ConfigurationSection server = section.getConfigurationSection(id);
                if (server == null) continue;
                String iconId = server.getString("icon", "");
                Glyph icon = Glyph.fromId(iconId);
                if (icon == null) {
                    logger.warning("Unknown icon '" + iconId + "' for server " + id + ", using " + Glyph.ICON_STAR.id());
                    icon = Glyph.ICON_STAR;
                }
                servers.add(new ServerEntry(id, server.getString("display-name", id), icon,
                        List.copyOf(server.getStringList("description"))));
            }
        }
        return new SelectorConfig(root.getString("title", DEFAULT_TITLE), List.copyOf(servers));
    }
}
```

```java
        live.put(key(id), new Live(current.status(), current.label(), players));
```

```java
import com.google.common.io.ByteArrayDataOutput;
import com.google.common.io.ByteStreams;
import org.bukkit.Bukkit;
import org.bukkit.entity.Player;

import us.ajg0702.queue.api.spigot.AjQueueSpigotAPI;

import java.util.List;
import java.util.concurrent.TimeUnit;

/** Keeps the registry's player counts (BungeeCord channel) and statuses (ajQueue) fresh. */
public final class ProxyPoller {

    private static final long INTERVAL_TICKS = 100;
    private static final long TIMEOUT_MS = 3000;

    private final Plugin plugin;
    private final ServerRegistry registry;

    public ProxyPoller(Plugin plugin, ServerRegistry registry) {
        this.plugin = plugin;
        this.registry = registry;
    }

    public void start() {
        Bukkit.getScheduler().runTaskTimer(plugin, this::poll, 20L, INTERVAL_TICKS);
    }

    /** Handles a message on the BungeeCord channel. Only PlayerCount replies are used. */
    public void handle(byte[] message) {
        ByteArrayDataInput in = ByteStreams.newDataInput(message);
        if (!"PlayerCount".equals(in.readUTF())) return;
        String server = in.readUTF();
        int count = in.readInt();
        registry.updatePlayers(server, count);
    }

    private void poll() {
        // Plugin messages and ajQueue's API both travel over a player's connection
        Player carrier = Bukkit.getOnlinePlayers().stream().findFirst().orElse(null);
        if (carrier == null) return;

        List<String> ids = registry.servers().stream().map(ServerEntry::id).toList();
        for (String id : ids) {
            ByteArrayDataOutput out = ByteStreams.newDataOutput();
            out.writeUTF("PlayerCount");
            out.writeUTF(id);
            carrier.sendPluginMessage(plugin, "BungeeCord", out.toByteArray());
        }

        Bukkit.getScheduler().runTaskAsynchronously(plugin, () -> {
            for (String id : ids) {
                ServerStatus status;
                String label;
                try {
                    String raw = AjQueueSpigotAPI.getInstance().getServerStatusString(id).get(TIMEOUT_MS, TimeUnit.MILLISECONDS);
                    status = ServerStatus.fromAjQueue(raw);
                    label = ServerStatus.plainText(raw);
                } catch (Exception e) {
                    status = ServerStatus.OFFLINE;
                    label = "Offline";
                }
                ServerStatus result = status;
                String resultLabel = label;
                Bukkit.getScheduler().runTask(plugin, () -> registry.updateStatus(id, result, resultLabel));
            }
        });
    }
}
```

```java
import java.util.concurrent.Future;
import java.util.concurrent.TimeUnit;
import java.util.function.Consumer;

/** Joins, leaves and reads ajQueue queues for a player. */
public final class QueueTracker {

    private static final long TIMEOUT_MS = 2000;

    private final Plugin plugin;

    public QueueTracker(Plugin plugin) {
        this.plugin = plugin;
    }

    public void join(Player player, ServerEntry server) {
        AjQueueSpigotAPI.getInstance().sudoQueue(player.getUniqueId(), server.id());
        Bukkit.getPluginManager().callEvent(new QueueJoinEvent(player, server.id()));
    }

    public void leave(Player player, String serverId) {
        // ajQueue has no API for leaving; its backend command forwards to the proxy
        player.performCommand("leavequeue " + serverId);
        Bukkit.getPluginManager().callEvent(new QueueLeaveEvent(player));
    }

    /** Reads the player's queue state off the main thread and hands it back on the main thread. */
    public void poll(Player player, Consumer<QueueState> callback) {
        UUID id = player.getUniqueId();
        Bukkit.getScheduler().runTaskAsynchronously(plugin, () -> {
            QueueState state;
            try {
                AjQueueSpigotAPI api = AjQueueSpigotAPI.getInstance();
                if (!Boolean.TRUE.equals(await(api.isInQueue(id)))) {
                    state = QueueState.NOT_QUEUED;
                } else {
                    state = QueueState.queued(
                            response(api.getRawQueueName(id), null),
                            response(api.getPosition(id), 0),
                            response(api.getTotalPositions(id), 0));
                }
            } catch (Exception e) {
                state = QueueState.UNAVAILABLE;
            }
            QueueState result = state;
            Bukkit.getScheduler().runTask(plugin, () -> callback.accept(result));
        });
    }

    private static <T> T await(Future<T> future) throws Exception {
        return future.get(TIMEOUT_MS, TimeUnit.MILLISECONDS);
    }

    private static <T> T response(Future<MessagedResponse<T>> future, T fallback) throws Exception {
        MessagedResponse<T> response = await(future);
        return response == null || response.getResponse() == null ? fallback : response.getResponse();
    }
}
```

```java
package lt.tomexas.serverselector.selector.ui;

import org.bukkit.Location;
import org.bukkit.util.Vector;

/**
 * A plane facing the camera. UI coordinates are blocks on the plane at {@link UiMath#REFERENCE_DISTANCE},
 * x to the right and y up from its centre.
 */
public final class UiPlane {

    private final Location eye;
    private final Vector forward;
    private final Vector right;
    private final Vector up;

    public UiPlane(Location eye) {
        this.eye = eye.clone();
        this.forward = eye.getDirection().normalize();
        this.right = forward.clone().crossProduct(new Vector(0, 1, 0)).normalize();
        this.up = right.clone().crossProduct(forward).normalize();
    }

    /** World position of (x, y), drawn at the given distance from the eye on the same line of sight. */
    public Location point(double x, double y, double depth) {
        double k = depth / UiMath.REFERENCE_DISTANCE;
        return eye.clone()
                .add(forward.clone().multiply(depth))
                .add(right.clone().multiply(x * k))
                .add(up.clone().multiply(y * k));
    }

    public float yaw() {
        return eye.getYaw();
    }
}
```

```java
import net.kyori.adventure.text.Component;
import org.bukkit.Bukkit;
import org.bukkit.Color;

/**
 * One text display only its viewer can see, positioned by its centre on a {@link UiPlane}.
 * Positions and scales are in reference-plane units, so an element looks the same size at any depth.
 */
public final class UiElement {

    /** Height of one text display line in font pixels. The display's origin is the bottom of its text. */
    static final double LINE_PX = 10.0;
    private static final float GROW_FROM = 0.85f;
    private static final int GROW_TICKS = 4;

    private final UiPlane plane;
    private final TextDisplay display;
    private final double depth;
    private final int lines;
    private double x;
    private double y;
    private float scaleX;
    private float scaleY;
    private boolean visible = true;

    private UiElement(UiPlane plane, TextDisplay display, double depth, int lines, double x, double y, float scale) {
        this.plane = plane;
        this.display = display;
        this.depth = depth;
        this.lines = lines;
        this.x = x;
        this.y = y;
        this.scaleX = scale;
        this.scaleY = scale;
    }

    public static UiElement spawn(Plugin plugin, Player viewer, UiPlane plane, Component text, int lines,
                                  double x, double y, double depth, float scale) {
        float start = scale * GROW_FROM * (float) (depth / UiMath.REFERENCE_DISTANCE);
        Location at = plane.point(x, originY(y, lines, scale), depth);
        TextDisplay display = at.getWorld().spawn(at, TextDisplay.class, d -> {
            d.setPersistent(false);
            d.setVisibleByDefault(false);
            d.setBillboard(Display.Billboard.CENTER);
            d.setBrightness(new Display.Brightness(15, 15));
            d.setBackgroundColor(Color.fromARGB(0, 0, 0, 0));
            d.setLineWidth(1000);
            d.setShadowed(false);
            d.text(text);
            d.setTransformation(transformation(start, start));
            viewer.showEntity(plugin, d);
        });
        UiElement element = new UiElement(plane, display, depth, lines, x, y, scale);
        // Grow to full size once the client has the entity, which reads as the UI opening up
        Bukkit.getScheduler().runTaskLater(plugin, () -> element.applyScale(GROW_TICKS), 2L);
        return element;
    }

    public static UiElement glyph(Plugin plugin, Player viewer, UiPlane plane, Glyph glyph,
                                  double x, double y, double depth, float scale) {
        return spawn(plugin, viewer, plane, glyph.component(), 1, x, y, depth, scale);
    }

    public void moveTo(double x, double y, float scale, int ticks) {
        layout(x, y, scale, scale, ticks);
    }

    public void layout(double x, double y, float scaleX, float scaleY, int ticks) {
        boolean moved = x != this.x || y != this.y || scaleY != this.scaleY;
        boolean scaled = scaleX != this.scaleX || scaleY != this.scaleY;
        this.x = x;
        this.y = y;
        this.scaleX = scaleX;
        this.scaleY = scaleY;
        if (!display.isValid()) return;
        if (moved) {
            display.setTeleportDuration(Math.max(0, Math.min(59, ticks)));
            display.teleport(plane.point(x, originY(y, lines, scaleY), depth));
        }
        if (scaled) applyScale(ticks);
    }

    public void text(Component text) {
        display.text(text);
    }

    public void opacity(int alpha) {
        display.setTextOpacity((byte) Math.max(0, Math.min(255, alpha)));
    }

    public void shadow(boolean shadow) {
        display.setShadowed(shadow);
    }

    public void visible(boolean visible) {
        if (this.visible == visible) return;
        this.visible = visible;
        applyScale(0);
    }

    public void remove() {
        display.remove();
    }

    private void applyScale(int ticks) {
        if (!display.isValid()) return;
        float factor = (float) (depth / UiMath.REFERENCE_DISTANCE);
        display.setInterpolationDelay(0);
        display.setInterpolationDuration(ticks);
        display.setTransformation(visible ? transformation(scaleX * factor, scaleY * factor) : transformation(0, 0));
    }

    private static double originY(double centerY, int lines, float scale) {
        return centerY - lines * LINE_PX / 2 * UiMath.PIXEL * scale;
    }

    private static Transformation transformation(float scaleX, float scaleY) {
        return new Transformation(new Vector3f(), new AxisAngle4f(), new Vector3f(scaleX, scaleY, scaleX), new AxisAngle4f());
    }
}
```

```java
    private final List<Card> cards = new ArrayList<>();
    private final UiElement leftArrow;
    private final UiElement rightArrow;
    private int selected;
    private int hovered = NONE;
    private boolean dimmed;
    private int shakeTicks;
    private int ticks;

    public CarouselScreen(Plugin plugin, Player player, UiPlane plane, ServerRegistry registry, int selected,
                          Consumer<ServerEntry> onJoin) {
        this.plugin = plugin;
        this.player = player;
        this.plane = plane;
        this.registry = registry;
        this.onJoin = onJoin;

        List<ServerEntry> servers = registry.servers();
        this.selected = servers.isEmpty() ? 0 : CarouselLayout.wrap(selected, servers.size());
        for (int i = 0; i < servers.size(); i++) {
            cards.add(new Card(servers.get(i), CarouselLayout.slot(i, this.selected, servers.size())));
        }

        leftArrow = UiElement.glyph(plugin, player, plane, Glyph.ARROW_LEFT, -ARROW_X, CARD_Y, DEPTH, ARROW_SCALE);

    @Override

            shakeTicks--;

            cards.get(selected).place(1, offset);

    @Override

        restyle(previous);
        restyle(target);
        if (target != NONE) player.playSound(player, Sound.BLOCK_NOTE_BLOCK_HAT, 0.3f, 1.6f);

        rightArrow.moveTo(ARROW_X, CARD_Y, ARROW_SCALE, HOVER_TICKS);

        leftArrow.remove();

    private void rotate(int steps) {

        selected = CarouselLayout.wrap(selected + steps, cards.size());
        hovered = NONE;
        for (int i = 0; i < cards.size(); i++) {
            Card card = cards.get(i);
            card.slot = CarouselLayout.slot(i, selected, cards.size());
            card.place(SLIDE_TICKS, 0);

        player.playSound(player, Sound.ITEM_BOOK_PAGE_TURN, 0.6f, 1.2f);

    private void join() {
        Card card = cards.get(selected);
        if (registry.live(card.server.id()).status().joinable()) {
            player.playSound(player, Sound.UI_BUTTON_CLICK, 0.6f, 1.0f);
            onJoin.accept(card.server);
        } else {
            shakeTicks = SHAKE_TICKS;
            player.playSound(player, Sound.ENTITY_VILLAGER_NO, 0.6f, 1.0f);
        }
    }

    private int targetAt(double x, double y) {
        if (cards.size() > 1) {
            if (new Rect(-ARROW_X, CARD_Y, ARROW_SIZE, ARROW_SIZE).contains(x, y)) return LEFT_ARROW;
            if (new Rect(ARROW_X, CARD_Y, ARROW_SIZE, ARROW_SIZE).contains(x, y)) return RIGHT_ARROW;
        }
        if (cards.isEmpty()) return NONE;
        // The centre card is drawn over the side cards, so it wins where they overlap
        if (cards.get(selected).rect().contains(x, y)) return selected;
        for (int i = 0; i < cards.size(); i++) {
            Card card = cards.get(i);
            if (i != selected && CarouselLayout.visible(card.slot) && card.rect().contains(x, y)) return i;
        }
        return NONE;
    }

    private void restyle(int target) {
        if (target == LEFT_ARROW) {
            leftArrow.moveTo(-ARROW_X, CARD_Y, hovered == LEFT_ARROW ? ARROW_HOVER_SCALE : ARROW_SCALE, HOVER_TICKS);
        } else if (target == RIGHT_ARROW) {
            rightArrow.moveTo(ARROW_X, CARD_Y, hovered == RIGHT_ARROW ? ARROW_HOVER_SCALE : ARROW_SCALE, HOVER_TICKS);
        } else if (target >= 0) {
            cards.get(target).place(HOVER_TICKS, 0);
        }
    }

    private Component statusLine(ServerEntry server) {
        ServerRegistry.Live live = registry.live(server.id());
        NamedTextColor color = switch (live.status()) {
            case ONLINE -> NamedTextColor.GREEN;
            case LIMITED -> NamedTextColor.YELLOW;
            case OFFLINE, UNKNOWN -> NamedTextColor.RED;
        };
        String label = live.status() == ServerStatus.UNKNOWN ? "Checking..." : live.label();
        Component line = Component.text("\u25CF " + label, color);
        if (live.status().joinable()) {
            line = line.append(Component.text("  \u00B7  " + live.players() + " playing", NamedTextColor.GRAY));
        }
        return line;
    }

    private final class Card {

        private final ServerEntry server;
        private final UiElement background;
        private final UiElement icon;
        private final UiElement name;
        private final UiElement status;
        private final UiElement description;
        private Slot slot;

        private Card(ServerEntry server, Slot slot) {
            this.server = server;
            this.slot = slot;
            double x = CarouselLayout.x(slot);
            float s = CarouselLayout.scale(slot);
            background = UiElement.glyph(plugin, player, plane, slot == Slot.CENTER ? Glyph.CARD_SELECTED : Glyph.CARD,
                    x, CARD_Y, DEPTH, CARD_SCALE * s);
            icon = UiElement.spawn(plugin, player, plane, server.icon().component(), 1,
                    x, CARD_Y + ICON_Y * s, CONTENT_DEPTH, ICON_SCALE * s);
            name = UiElement.spawn(plugin, player, plane,
                    Component.text(server.displayName(), NamedTextColor.WHITE, TextDecoration.BOLD), 1,
                    x, CARD_Y + NAME_Y * s, CONTENT_DEPTH, NAME_SCALE * s);
            status = UiElement.spawn(plugin, player, plane, statusLine(server), 1,
                    x, CARD_Y + STATUS_Y * s, CONTENT_DEPTH, STATUS_SCALE * s);
            description = UiElement.spawn(plugin, player, plane,
                    Component.text(String.join("\n", server.description()), NamedTextColor.GRAY),
                    Math.max(1, server.description().size()),
                    x, CARD_Y + DESCRIPTION_Y * s, CONTENT_DEPTH, DESCRIPTION_SCALE * s);
            name.shadow(true);
            status.shadow(true);
            description.shadow(true);
            place(0, 0);
        }

        private void place(int ticks, double shake) {
            boolean center = slot == Slot.CENTER;
            boolean shown = CarouselLayout.visible(slot);
            boolean hover = !dimmed && hovered == cards.indexOf(this);
            float s = CarouselLayout.scale(slot) * (hover ? (center ? CENTER_HOVER_GROW : SIDE_HOVER_GROW) : 1f);
            double x = CarouselLayout.x(slot) + shake;

            background.text((center ? Glyph.CARD_SELECTED : Glyph.CARD).component());
            background.moveTo(x, CARD_Y, CARD_SCALE * s, ticks);
            icon.moveTo(x, CARD_Y + ICON_Y * s, ICON_SCALE * s, ticks);
            name.moveTo(x, CARD_Y + NAME_Y * s, NAME_SCALE * s, ticks);
            status.moveTo(x, CARD_Y + STATUS_Y * s, STATUS_SCALE * s, ticks);
            description.moveTo(x, CARD_Y + DESCRIPTION_Y * s, DESCRIPTION_SCALE * s, ticks);

            background.visible(shown);
            icon.visible(shown);
            name.visible(shown);
            status.visible(shown && center);
            description.visible(shown && center);

            int alpha = dimmed ? DIM_ALPHA : (center || hover ? 255 : SIDE_ALPHA);
            for (UiElement element : List.of(background, icon, name, status, description)) element.opacity(alpha);
        }

        private void refreshStatus() {
            status.text(statusLine(server));
        }

        private Rect rect() {
            float s = CarouselLayout.scale(slot);
            return new Rect(CarouselLayout.x(slot), CARD_Y, CARD_WIDTH * s, CARD_HEIGHT * s);
        }

        private void remove() {
            for (UiElement element : List.of(background, icon, name, status, description)) element.remove();
        }
    }
}
```

```java
/** One player's open selector: seat, camera, cursor and the active screens. */
final class SelectorSession {

    private static final long CLICK_DELAY_MS = 250;
    // ajQueue may not report the new queue straight away; don't close the panel in the meantime
    private static final long JOIN_GRACE_MS = 3000;
    private static final double BOUNDS_X = 2.2;
    private static final double BOUNDS_Y = 1.25;
    private static final double CURSOR_DEPTH = 1.8;
    private static final float CURSOR_SCALE = 1.0f;          // 16 px glyph -> 0.4
    // The cursor glyph's tip is its top-left corner, so its centre sits down and to the right of the cursor point
    private static final double CURSOR_TIP_X = 0.2;
    private static final double CURSOR_TIP_Y = -0.2;
    private static final double VIGNETTE_DEPTH = 2.4;
    private static final float VIGNETTE_SCALE = 1.6f;        // 256x144 px -> 10.2 x 5.8, wider than any FOV
    private static final double TITLE_Y = 1.15;
    private static final float TITLE_SCALE = 1.4f;
    private static final int QUEUE_POLL_TICKS = 20;

    private final Plugin plugin;
    private final Player player;
    private final ServerRegistry registry;
    private final QueueTracker queue;
    private final UiPlane plane;
    private final Pig seat;
    private final UiElement vignette;
    private final UiElement title;
    private final CarouselScreen carousel;
    private final UiElement cursor;
    private QueuePanel queuePanel;
    private ServerEntry queuedFor;
    private double cursorX;
    private double cursorY;
    private boolean pointing;
    private long lastClick;
    private long joinPendingUntil;
    private int ticks;
    private boolean closed;

    SelectorSession(Plugin plugin, Player player, ArmorStand camera, SelectorConfig config, ServerRegistry registry,
                    QueueTracker queue, int selected) {
        this.plugin = plugin;
        this.player = player;
        this.registry = registry;
        this.queue = queue;

        Location eye = camera.getEyeLocation();
        this.plane = new UiPlane(eye);
        this.seat = spawnSeat(eye);

        // Face the player the same way as the camera with a level head, so the cursor starts in the middle
        Location start = seat.getLocation();
        start.setYaw(eye.getYaw());
        start.setPitch(0f);
        player.teleport(start);
        seat.addPassenger(player);
        lockCamera();

        vignette = UiElement.glyph(plugin, player, plane, Glyph.VIGNETTE, 0, 0, VIGNETTE_DEPTH, VIGNETTE_SCALE);
        title = UiElement.spawn(plugin, player, plane,
                Component.text(config.title(), NamedTextColor.GOLD, TextDecoration.BOLD), 1,
                0, TITLE_Y, UiMath.REFERENCE_DISTANCE, TITLE_SCALE);
        title.shadow(true);
        carousel = new CarouselScreen(plugin, player, plane, registry, selected, this::joinQueue);
        cursor = UiElement.glyph(plugin, player, plane, Glyph.CURSOR_ARROW,
                CURSOR_TIP_X * CURSOR_SCALE, CURSOR_TIP_Y * CURSOR_SCALE, CURSOR_DEPTH, CURSOR_SCALE);
    }

    void tick() {
        if (closed) return;
        Location location = player.getLocation();
        double x = UiMath.clamp(UiMath.project(UiMath.wrapDegrees(location.getYaw() - plane.yaw())), -BOUNDS_X, BOUNDS_X);
        // Pitch is positive when looking down
        double y = UiMath.clamp(UiMath.project(-location.getPitch()), -BOUNDS_Y, BOUNDS_Y);
        if (x != cursorX || y != cursorY) {
            cursorX = x;
            cursorY = y;
            cursor.moveTo(x + CURSOR_TIP_X * CURSOR_SCALE, y + CURSOR_TIP_Y * CURSOR_SCALE, CURSOR_SCALE, 1);
        }

        carousel.tick();
        if (queuePanel != null) queuePanel.tick();
        boolean clickable = activeScreen().hover(cursorX, cursorY);
        if (clickable != pointing) {
            pointing = clickable;
            cursor.text((clickable ? Glyph.CURSOR_HAND : Glyph.CURSOR_ARROW).component());
        }

        if (++ticks % QUEUE_POLL_TICKS == 0) queue.poll(player, this::onQueueState);
    }

    void click() {
        long now = System.currentTimeMillis();
        if (now - lastClick < CLICK_DELAY_MS) return;
        lastClick = now;
        activeScreen().click(cursorX, cursorY);
    }

    void scroll(int direction) {
        if (queuePanel == null) carousel.scroll(direction);
    }

    int selectedIndex() {
        return carousel.selectedIndex();
    }

    void close() {
        closed = true;
        if (queuePanel != null) queuePanel.close();
        carousel.close();
        vignette.remove();
        title.remove();
        cursor.remove();
        seat.eject();
        seat.remove();
    }

    private Pig spawnSeat(Location eye) {
        // The pig is both seat and camera: a riding client keeps sending its head rotation, and only keeps its
        // camera on its own vehicle. Its eyes go where the armor stand's are, so the view is unchanged.
        Pig pig = eye.getWorld().spawn(eye, Pig.class, p -> {
            p.setInvisible(true);
            p.setAI(false);
            p.setGravity(false);
            p.setSilent(true);
            p.setInvulnerable(true);
            p.setCollidable(false);
            p.setPersistent(false);
        });
        pig.teleport(eye.clone().subtract(0, pig.getEyeHeight(), 0));
        pig.setRotation(eye.getYaw(), eye.getPitch());
        return pig;
    }

    private void lockCamera() {
        // The client ignores a camera packet for an entity it has not been sent yet, so wait for the pig
        Bukkit.getScheduler().runTaskLater(plugin, () -> {
            if (closed || !player.isOnline() || !seat.isValid()) return;
            PacketContainer packet = ProtocolLibrary.getProtocolManager().createPacket(PacketType.Play.Server.CAMERA);
            packet.getIntegers().write(0, seat.getEntityId());
            ProtocolLibrary.getProtocolManager().sendServerPacket(player, packet);
        }, 2L);
    }

    private Screen activeScreen() {
        return queuePanel != null ? queuePanel : carousel;
    }

    private void joinQueue(ServerEntry server) {
        queue.join(player, server);
        joinPendingUntil = System.currentTimeMillis() + JOIN_GRACE_MS;
        openQueuePanel(server);
    }

    private void leaveQueue() {
        if (queuedFor != null) queue.leave(player, queuedFor.id());
        joinPendingUntil = 0;
        closeQueuePanel();
    }

    private void onQueueState(QueueState state) {
        if (closed) return;
        if (!state.available()) {
            if (queuePanel != null) queuePanel.showUnavailable();
            return;
        }
        if (state.inQueue()) {
            ServerEntry server = registry.entry(state.server());
            if (server == null && state.server() != null) {
                server = new ServerEntry(state.server(), state.server(), Glyph.ICON_STAR, List.of());
            }
            if (server == null) server = queuedFor;
            if (server == null) return;
            if (queuePanel == null || !server.id().equalsIgnoreCase(queuedFor.id())) openQueuePanel(server);
            queuePanel.update(state);
        } else if (queuePanel != null && System.currentTimeMillis() > joinPendingUntil) {
            closeQueuePanel();
        }
    }

    private void openQueuePanel(ServerEntry server) {
        if (queuePanel != null) queuePanel.close();
        queuedFor = server;
        carousel.dim(true);
        queuePanel = new QueuePanel(plugin, player, plane, server, this::leaveQueue);
    }

    private void closeQueuePanel() {
        if (queuePanel == null) return;
        queuePanel.close();
        queuePanel = null;
        queuedFor = null;
        carousel.dim(false);
    }
}
```

```java
package lt.tomexas.serverselector.selector;

import lt.tomexas.serverselector.queue.QueueTracker;
import lt.tomexas.serverselector.server.SelectorConfig;
import lt.tomexas.serverselector.server.ServerRegistry;
import org.bukkit.Bukkit;

import org.bukkit.entity.Player;
import org.bukkit.plugin.Plugin;

import java.util.ArrayList;
import java.util.HashMap;
import java.util.Map;
import java.util.UUID;
import java.util.function.Supplier;

/** Opens and closes selector sessions and drives them every tick. */
public final class SelectorManager {

    private final Plugin plugin;
    private final Supplier<ArmorStand> camera;
    private final SelectorConfig config;
    private final ServerRegistry registry;
    private final QueueTracker queue;
    private final Map<UUID, SelectorSession> sessions = new HashMap<>();
    // Which server was in the middle when the player last closed the selector
    private final Map<UUID, Integer> lastSelected = new HashMap<>();
    private boolean warnedNoCamera;

    public SelectorManager(Plugin plugin, Supplier<ArmorStand> camera, SelectorConfig config, ServerRegistry registry,
                           QueueTracker queue) {
        this.plugin = plugin;
        this.camera = camera;
        this.config = config;
        this.registry = registry;
        this.queue = queue;
        Bukkit.getScheduler().runTaskTimer(plugin, this::tick, 1L, 1L);
    }

    public void open(Player player) {
        close(player);
        ArmorStand stand = camera.get();
        if (stand == null || !stand.isValid()) {
            if (!warnedNoCamera) {
                plugin.getLogger().warning("Camera armor stand is missing or not loaded, the selector is disabled.");
                warnedNoCamera = true;
            }
            return;
        }
        sessions.put(player.getUniqueId(), new SelectorSession(plugin, player, stand, config, registry, queue,
                lastSelected.getOrDefault(player.getUniqueId(), 0)));
    }

    public void close(Player player) {
        // Removed from the map first so the dismount listener lets the player off the seat
        SelectorSession session = sessions.remove(player.getUniqueId());
        if (session == null) return;
        lastSelected.put(player.getUniqueId(), session.selectedIndex());
        session.close();
    }

    public void closeAll() {
        new ArrayList<>(sessions.values()).forEach(SelectorSession::close);
        sessions.clear();
    }

    public boolean isAttached(Player player) {
        return sessions.containsKey(player.getUniqueId());
    }

    public void click(Player player) {
        SelectorSession session = sessions.get(player.getUniqueId());
        if (session != null) session.click();
    }

    public void scroll(Player player, int direction) {
        SelectorSession session = sessions.get(player.getUniqueId());
        if (session != null) session.scroll(direction);
    }

    private void tick() {
        new ArrayList<>(sessions.values()).forEach(SelectorSession::tick);
    }
}
```

```java
import org.bukkit.plugin.messaging.PluginMessageListener;
import org.jetbrains.annotations.NotNull;

import java.sql.SQLException;
import java.util.UUID;
import java.util.logging.Logger;

public final class Main extends JavaPlugin implements PluginMessageListener {

    private static Main instance;
    private ProtocolManager protocolManager;
    private Database database;
    private ArmorStand armorStand;
    private SelectorManager selectorManager;
    private ProxyPoller proxyPoller;

    @Override
    public void onEnable() {
        instance = this;
        saveDefaultConfig();
        protocolManager = ProtocolLibrary.getProtocolManager();

        this.setupDatabase();

        boolean nexo = getServer().getPluginManager().isPluginEnabled("Nexo");
        Glyph.setNexoAvailable(nexo);
        if (!nexo) getLogger().info("Nexo not found, the selector uses plain-text glyphs.");

        SelectorConfig config = SelectorConfig.from(getConfig(), getLogger());
        ServerRegistry registry = new ServerRegistry(config.servers());
        selectorManager = new SelectorManager(this, this::getArmorStand, config, registry, new QueueTracker(this));
        proxyPoller = new ProxyPoller(this, registry);
        proxyPoller.start();

        PluginManager pluginManager = getServer().getPluginManager();
        pluginManager.registerEvents(new PlayerJoinListener(), this);
        pluginManager.registerEvents(new PlayerQuitListener(), this);
        pluginManager.registerEvents(new PlayerToggleSneakListener(), this);
        pluginManager.registerEvents(new PlayerItemHeldListener(), this);
        pluginManager.registerEvents(new PlayerArmSwingListener(), this);
        pluginManager.registerEvents(new PlayerDismountListener(), this);
        pluginManager.registerEvents(new PlayerKickListener(), this);

        this.getServer().getMessenger().registerOutgoingPluginChannel(this, "BungeeCord");
        this.getServer().getMessenger().registerIncomingPluginChannel(this, "BungeeCord", this);
    }

    @Override
    public void onPluginMessageReceived(@NotNull String channel, @NotNull Player player, byte @NotNull [] message) {
        if (channel.equals("BungeeCord")) proxyPoller.handle(message);
    }

    @Override
    public void onDisable() {
        if (selectorManager != null) selectorManager.closeAll();
        if (protocolManager != null) protocolManager.removePacketListeners(this);
        try {
            if (database != null) database.closeConnection();
        } catch (SQLException e) {
            Logger.getLogger("Minecraft").severe("Failed to close the database connection! " + e.getMessage());
        }
    }

    private void setupDatabase() {
        try {
            if (!getDataFolder().exists()) {
                if (!getDataFolder().mkdirs()) {
                    getLogger().severe("Failed to create data folder!");
                    Bukkit.getPluginManager().disablePlugin(this);
                    return;
                }
            }
            database = new Database(getDataFolder().getAbsolutePath() + "/main.db");
        } catch (SQLException e) {
            getLogger().severe("Failed to connect to the database! " + e.getMessage());
            Bukkit.getPluginManager().disablePlugin(this);
        }
    }

    public static Main getInstance() {
        return instance;
    }

    public ProtocolManager getProtocolManager() {
        return protocolManager;
    }

    /** The camera armor stand, looked up again if it was not loaded yet when the plugin enabled. */
    public ArmorStand getArmorStand() {
        if ((armorStand == null || !armorStand.isValid()) && database != null && database.getUUID() != null) {
            Entity entity = Bukkit.getEntity(UUID.fromString(database.getUUID()));
            armorStand = entity instanceof ArmorStand stand ? stand : null;
        }
        return armorStand;
    }

    public SelectorManager getSelectorManager() {
        return selectorManager;
    }
}
```

```java
            player.addPotionEffect(new PotionEffect(PotionEffectType.INVISIBILITY, PotionEffect.INFINITE_DURATION, 1, false, false));

    }
```

```java
package lt.tomexas.serverselector.Listeners;

import lt.tomexas.serverselector.Main;
import org.bukkit.entity.Player;
import org.bukkit.event.EventHandler;
import org.bukkit.event.Listener;
import org.bukkit.event.player.PlayerItemHeldEvent;

public class PlayerItemHeldListener implements Listener {

    private final Main plugin = Main.getInstance();

    // Scrolling the hotbar rotates the carousel
    @EventHandler
    public void onPlayerItemHeld(PlayerItemHeldEvent event) {
        Player player = event.getPlayer();
        if (!plugin.getSelectorManager().isAttached(player)) return;
        event.setCancelled(true);

        int step = event.getNewSlot() - event.getPreviousSlot();
        // Scrolling past slot 8 wraps to 0 and the other way round
        if (step > 4) step = -1;
        else if (step < -4) step = 1;
        plugin.getSelectorManager().scroll(player, Integer.signum(step));
    }
}
```
